cmake_minimum_required(VERSION 3.24)
project(nect VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(NECT_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
option(NECT_DESKTOP "Build the Qt desktop authoring client" OFF)

find_path(BOOST_INCLUDE_DIR boost/json/src.hpp REQUIRED)

add_library(nect_core src/semantic_controls.cpp src/core.cpp src/blend.cpp src/arithmetic_blend.cpp src/shape.cpp src/text.cpp src/color.cpp src/history.cpp src/transform.cpp src/expression.cpp src/compositing.cpp src/offset.cpp src/raster.cpp)
target_sources(nect_core PRIVATE src/analysis_adoption.cpp)
target_include_directories(nect_core PUBLIC include PRIVATE ${BOOST_INCLUDE_DIR})
if(WIN32)
  target_link_libraries(nect_core PRIVATE dwrite windowscodecs bcrypt ole32)
endif()

add_library(nect_io src/io.cpp src/compatibility.cpp src/boost_json.cpp)
target_include_directories(nect_io PRIVATE ${BOOST_INCLUDE_DIR})
target_compile_definitions(nect_io PRIVATE BOOST_ALL_NO_LIB)
target_link_libraries(nect_io PUBLIC nect_core)

add_executable(nect src/main.cpp)
target_link_libraries(nect PRIVATE nect_io)

foreach(t nect_core nect_io nect)
  if(MSVC)
    target_compile_options(${t} PRIVATE /W4 /permissive- /utf-8)
  else()
    target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  endif()
  if(NECT_SANITIZE AND NOT MSVC)
    target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${t} PUBLIC -fsanitize=address,undefined)
  endif()
endforeach()
