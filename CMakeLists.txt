cmake_minimum_required(VERSION 3.24)
project(nect VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(NECT_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
option(NECT_DESKTOP "Build the Qt desktop authoring client" OFF)

find_path(BOOST_INCLUDE_DIR boost/json/src.hpp REQUIRED)

add_library(nect_core src/core.cpp)
target_include_directories(nect_core PUBLIC include)

add_library(nect_io src/io.cpp src/boost_json.cpp)
target_include_directories(nect_io PRIVATE ${BOOST_INCLUDE_DIR})
target_compile_definitions(nect_io PRIVATE BOOST_ALL_NO_LIB)
target_link_libraries(nect_io PUBLIC nect_core)

add_executable(nect src/main.cpp)
target_link_libraries(nect PRIVATE nect_io)

foreach(t nect_core nect_io nect)
  if(MSVC)
    target_compile_options(${t} PRIVATE /W4 /permissive- /utf-8)
  else()
    target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  endif()
  if(NECT_SANITIZE AND NOT MSVC)
    target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${t} PUBLIC -fsanitize=address,undefined)
  endif()
endforeach()

if(NECT_DESKTOP)
  find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets Network Test)
  add_library(nect_desktop_client src/desktop/host.cpp src/desktop/canvas.cpp src/desktop/window.cpp)
  target_include_directories(nect_desktop_client PUBLIC src/desktop)
  target_link_libraries(nect_desktop_client PUBLIC nect_io Qt6::Widgets Qt6::Network)
  add_executable(nect_desktop WIN32 src/desktop/main.cpp)
  target_link_libraries(nect_desktop PRIVATE nect_desktop_client)
  if(MSVC)
    target_compile_options(nect_desktop_client PRIVATE /W4 /permissive- /utf-8)
    target_compile_options(nect_desktop PRIVATE /W4 /permissive- /utf-8)
  endif()
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE nect_io)
  add_test(NAME core_contract COMMAND core_tests)
  if(NECT_DESKTOP)
    add_executable(desktop_host_tests tests/desktop_host_tests.cpp)
    target_link_libraries(desktop_host_tests PRIVATE nect_desktop_client)
    add_test(NAME desktop_host_contract COMMAND desktop_host_tests)
    add_executable(canvas_tests tests/canvas_tests.cpp)
    target_link_libraries(canvas_tests PRIVATE nect_desktop_client Qt6::Test)
    add_test(NAME canvas_interaction COMMAND canvas_tests)
    add_executable(window_tests tests/window_tests.cpp)
    target_link_libraries(window_tests PRIVATE nect_desktop_client Qt6::Test)
    add_test(NAME window_interaction COMMAND window_tests)
    add_executable(canvas_benchmark tests/canvas_benchmark.cpp)
    target_link_libraries(canvas_benchmark PRIVATE nect_desktop_client)
  endif()
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME process_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/process_tests.py $<TARGET_FILE:nect>)
  if(NECT_DESKTOP)
    add_test(NAME mcp_desktop_contract COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/mcp_desktop_tests.py $<TARGET_FILE:nect_desktop>)
    set_tests_properties(canvas_interaction window_interaction desktop_host_contract mcp_desktop_contract PROPERTIES TIMEOUT 60)
  endif()
endif()
